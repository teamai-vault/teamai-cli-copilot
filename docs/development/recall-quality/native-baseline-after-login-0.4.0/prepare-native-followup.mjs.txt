function listFiles(dir,pre=''){
 const out=[];
 for(const entry of fs.readdirSync(path.join(dir,pre),{withFileTypes:true}).sort((a,b)=>a.name.localeCompare(b.name))){
  const rel=path.join(pre,entry.name), st=fs.lstatSync(path.join(dir,rel));
  if(st.isSymbolicLink()) throw Error('Controlled source link refused');
  if(st.isDirectory()) out.push(...listFiles(dir,rel)); else if(st.isFile()) out.push(rel);
 }
 return out;
}
